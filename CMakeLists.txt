cmake_minimum_required(VERSION 3.22)
project(vesper VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VESPER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(VESPER_VULKAN "Build Vulkan compute backend when its SDK is available" ON)
option(VESPER_WORKBENCH "Build SDL workbench when its SDK is available" ON)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wstrict-prototypes)
    if(VESPER_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
enable_testing()

add_library(vesper STATIC src/geometry.c src/bvh.c)
target_include_directories(vesper PUBLIC include)
target_link_libraries(vesper PUBLIC m)
target_sources(vesper PRIVATE src/spectral.c)
include(cmake/observer.cmake)
target_sources(vesper PRIVATE src/scene.c)
target_sources(vesper PRIVATE src/scene_io.c)
target_sources(vesper PRIVATE src/scatter.c)
target_sources(vesper PRIVATE src/volume.c)
target_sources(vesper PRIVATE src/transport.c)
target_sources(vesper PRIVATE src/film.c)
target_sources(vesper PRIVATE src/checkpoint.c src/export.c)

add_executable(vesper-cli app/main.c)
set_target_properties(vesper-cli PROPERTIES OUTPUT_NAME vesper)
target_link_libraries(vesper-cli PRIVATE vesper)
install(TARGETS vesper vesper-cli)
install(DIRECTORY include/vesper DESTINATION include)

if(VESPER_VULKAN)
    find_package(Vulkan QUIET)
    find_program(GLSLANG glslangValidator)
endif()
if(VESPER_VULKAN AND Vulkan_FOUND AND GLSLANG)
    set(shader "${CMAKE_CURRENT_SOURCE_DIR}/shaders")
    set(spirv "${CMAKE_CURRENT_BINARY_DIR}/wavefront.spv")
    set(embed "${CMAKE_CURRENT_BINARY_DIR}/generated/wavefront.inc")
    add_custom_command(OUTPUT "${embed}"
        COMMAND "${GLSLANG}" --target-env vulkan1.1 "-I${shader}" -V "${shader}/wavefront.comp" -o "${spirv}"
        COMMAND "${CMAKE_COMMAND}" "-DINPUT=${spirv}" "-DOUTPUT=${embed}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_spirv.cmake"
        DEPENDS shaders/wavefront.comp shaders/layout.glsl shaders/geometry.glsl shaders/scatter.glsl shaders/volume.glsl shaders/texture.glsl cmake/embed_spirv.cmake
        VERBATIM)
    target_sources(vesper PRIVATE src/gpu.c "${embed}")
    target_link_libraries(vesper PUBLIC Vulkan::Vulkan)
else()
    target_sources(vesper PRIVATE src/gpu_stub.c)
endif()
target_sources(vesper PRIVATE src/assets.c)
target_sources(vesper PRIVATE src/inverse.c)
add_executable(vesper-inverse app/inverse_main.c)
target_link_libraries(vesper-inverse PRIVATE vesper)
install(TARGETS vesper-inverse)
add_executable(vesper-geometry-check tests/geometry.c)
target_link_libraries(vesper-geometry-check PRIVATE vesper)
add_executable(vesper-numerical-check tests/numerical.c)
target_link_libraries(vesper-numerical-check PRIVATE vesper)
add_test(NAME geometry COMMAND vesper-geometry-check)
add_test(NAME spectral COMMAND vesper-numerical-check spectral)
add_test(NAME scattering COMMAND vesper-numerical-check bsdf)
add_test(NAME volume COMMAND vesper-numerical-check volume)

if(VESPER_WORKBENCH)
    find_package(SDL2 2.0.22 QUIET CONFIG)
    if(TARGET SDL2::SDL2)
        add_executable(vesper-workbench app/workbench.c)
        target_link_libraries(vesper-workbench PRIVATE vesper SDL2::SDL2)
        install(TARGETS vesper-workbench)
    endif()
endif()
add_executable(vesper-temporal-check tests/temporal.c)
target_link_libraries(vesper-temporal-check PRIVATE vesper)
add_test(NAME temporal_mis COMMAND vesper-temporal-check)
add_executable(vesper-benchmark tests/benchmark.c)
target_link_libraries(vesper-benchmark PRIVATE vesper)
add_executable(vesper-inverse-check tests/inverse.c)
target_link_libraries(vesper-inverse-check PRIVATE vesper)
add_test(NAME inverse_derivatives COMMAND vesper-inverse-check)
add_executable(vesper-api-example examples/create_scene.c)
target_link_libraries(vesper-api-example PRIVATE vesper)
target_sources(vesper PRIVATE src/gltf.c)
target_include_directories(vesper SYSTEM PRIVATE vendor/cgltf)
target_sources(vesper PRIVATE src/texture.c)
install(FILES vendor/cgltf/LICENSE DESTINATION share/licenses/vesper RENAME cgltf-LICENSE)
install(FILES data/README.md DESTINATION share/vesper/observer)
